# Everything SynexAgent needs EXCEPT the ONNX Runtime package itself. Deliberately split out (see
# requirements.txt) so a Jetson deployment can install this core set first, run hardware detection
# (scripts/detect_jetson_env.py), and only THEN choose a single ONNX Runtime package (CPU or a
# Jetson-specific GPU build) matching the detected environment -- never a generic PyPI
# `onnxruntime` install before that decision is made, and never both a CPU and a GPU build
# installed at once.
fastapi==0.141.1
uvicorn==0.52.4
# numpy 2.5.x requires Python >=3.12; the backend/production CI jobs (and the documented
# Python 3.11 support target) run on 3.11, where 2.5.3 has no matching wheel and the install
# fails. 2.4.6 is the newest 2.4.x, is compatible with Python 3.11-3.13, and provides the same
# NumPy 2.x API this project uses.
numpy==2.4.6
pydantic==2.13.5
typing_extensions==4.16.0
httpx==0.28.1
pyjwt==2.10.1
cryptography==44.0.0
redis==8.1.0
# Postgres driver for PostgresNovaCaseRepository/PostgresAuditRepository (see
# services/nova_repository.py, services/audit.py) -- an opt-in production-grade backend selected
# via NOVA_POSTGRES_URL, same shape as the existing SYNEX_REDIS_URL opt-in for idempotency/smart
# launch. Always installed (like redis above) so the opt-in works without a separate deploy step;
# the binary wheel avoids needing libpq-dev to build from source.
psycopg[binary]==3.2.4
